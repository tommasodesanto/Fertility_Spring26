\section{Motivation}

\begin{frame}{Research question}
  \Large
  How do housing costs and access to family-sized space affect fertility when
  births also reshape future population and housing demand?

  \vspace{1.0em}
  \normalsize
  \begin{itemize}
    \item Children raise current demand for rooms when households have little liquid wealth.
    \item Tenure and down-payment constraints determine which households can obtain those rooms.
    \item Birth cohorts later become persons and household heads, feeding back into housing demand.
  \end{itemize}

  \takeaway{Housing is both a current input into family formation and a market whose future scale depends on family formation.}
\end{frame}

\begin{frame}{Contribution}
  \begin{enumerate}
    \item \textbf{Household mechanism:} sequential fertility, housing size, tenure, saving, income risk, and warm-glow bequests are chosen jointly.
    \item \textbf{Demographic equilibrium:} births create persons; survival, migration, and head formation create the household-head stock.
    \item \textbf{Transition equilibrium:} forward-looking asset prices, housing supply, and an equal property-tax rebate clear jointly with household and demographic dynamics.
  \end{enumerate}

  \vspace{0.8em}
  The active implementation is a single national housing market. Location is
  therefore degenerate here; spatial sorting is a separate extension.

  \takeaway{Separating persons from household heads changes both the transition law and the definition of a steady state.}
\end{frame}

\section{Mechanism}

\begin{frame}{The full cost of a child}
  In a compact model, a young household chooses adult housing $h$ and children
  $n$. One child requires $a$ units of space and $q$ units of other goods.
  Limited access to family housing adds a goods-equivalent wedge $z$:
  \[
    n(r,z)=\frac{v}{q+a(r+z)},
    \qquad
    d_Y(r,z)=\frac{\alpha}{r}+a\,n(r,z).
  \]

  \begin{itemize}
    \item $r$ is the current price of housing services.
    \item $z$ summarizes the rental-size limit and the down-payment constraint.
    \item A higher $r$ or $z$ raises the full resource cost $q+a(r+z)$ of a child.
  \end{itemize}

  \takeaway{Housing access acts like a household-specific addition to the price of children.}
  \source{Compact analytical model; the representative-family law is intuition, not the quantitative demographic closure.}
\end{frame}

\section{Lifecycle model}

\begin{frame}{Lifecycle timing and state}
  One period is four years. A household enters at age 18, can live through age
  85, and carries the beginning-of-period state
  \[
    x=(b,z,a,o,n,m).
  \]

  \begin{itemize}
    \item $b$: liquid net wealth; $z$: persistent earnings; $a$: age.
    \item $o$: renter or owner product; $n$: literal parity; $m$: number of dependent children at home.
    \item Fertility is sequential: a successful birth raises parity by one; children then mature independently.
    \item The active national specification has one location; no migration choice is assigned to households.
  \end{itemize}

  \takeaway{Children affect current housing needs immediately, but their effect on the number of household heads arrives through a separate demographic law.}
\end{frame}

\begin{frame}{Preferences and child space}
  Let $m$ be the number of dependent children currently at home. Flow utility is
  \[
    u_t(c,s;m)=e(m)^{\sigma-1}
    \frac{\left[c^{\alpha_0}\bigl(s-\bar h(m)\bigr)^{1-\alpha_0}\right]^{1-\sigma}}
    {1-\sigma}+\psi_t m,
  \]
  with
  \[
    e(m)=\left(\frac{2+0.7m}{2}\right)^{0.7},
    \qquad
    \bar h(m)=h_1\mathbf 1\{m>0\}+h_m m.
  \]

  \begin{itemize}
    \item $e(m)$ is a consumption equivalence scale.
    \item $h_1$ is a first-child room jump and $h_m$ is the per-child room floor.
    \item Owner services are $s=\chi H_k$; renter services are the chosen rental rooms.
  \end{itemize}
\end{frame}

\begin{frame}{Housing, finance, and bequests}
  \begin{itemize}
    \item Renters choose continuous rooms up to 6; owners choose $H_k\in\{2,4,6,8,10\}$.
    \item A purchase requires cash of at least $(1-\phi)P_tH_k$, with $\phi=0.80$.
    \item Owners pay depreciation and property tax and incur a proportional moving cost when changing products.
    \item New unsecured borrowing is ruled out; inherited unsecured debt is age-tapered, while owner debt also has a collateral floor.
  \end{itemize}

  Warm-glow bequest utility is
  \[
    \mathcal B(w^e)=\theta_0
    \frac{(\theta_1+\max\{w^e,0\})^{1-\sigma}}{1-\sigma}.
  \]

  Estates affect saving and old-age housing retention. They are not assigned
  to particular descendants; entrant wealth is externally distributed.
\end{frame}

\begin{frame}{Sequential fertility and the Bellman problem}
  At an eligible state, the household compares waiting with attempting the next
  birth. Let $\pi_a$ be age-$a$ conception success, let $F_1$ be the one-time
  first-birth cost, and let $x^+$ raise parity by one and add one dependent
  child. Before the attempt taste shock,
  \[
    W_t^0(x)\equiv\mathcal V_t(x),\qquad
    W_t^A(x)=\pi_a\!\left[\mathcal V_t(x^+)
      -F_1\mathbf 1\{n=0\}\right]+(1-\pi_a)\mathcal V_t(x),
  \]
  and the attempt probability is
  \[
    \Pr(d=1\mid x,t)=
    \frac{\exp\{W_t^A(x)/\kappa(n)\}}
    {\exp\{W_t^A(x)/\kappa(n)\}+\exp\{W_t^0(x)/\kappa(n)\}}.
  \]
  Here $\mathcal V_t$ already optimizes tenure, rooms, consumption, and saving
  conditional on the parity branch; $\kappa(0)=\kappa_1$ and
  $\kappa(n\ge1)=\kappa_C$.

  \takeaway{Housing affects fertility through current child-space needs, tenure feasibility, and the continuation value of wealth and housing.}
\end{frame}

\begin{frame}{Housing-market equilibrium}
  Household policies imply occupied housing demand $D_t$. Supply is static
  elastic around the 2023 anchor:
  \[
    H_t^s=H_{2023}\left(\frac{P_t}{P_{2023}}\right)^{\eta},
    \qquad \eta=1.75,
    \qquad D_t=H_t^s.
  \]

  Asset prices and rents satisfy the one-period no-arbitrage identity
  \[
    r_t+P_{t+1}=(R+\delta+\tau_H)P_t.
  \]

  \begin{itemize}
    \item $P_t$ capitalizes the next price; $r_t$ is the current renter price.
    \item $R$ is the four-year gross liquid return; $\delta$ and $\tau_H$ are four-year wedges.
    \item Households solve backward along the price and transfer paths.
    \item The distribution, births, persons, and heads move forward.
  \end{itemize}
\end{frame}

\begin{frame}{The person-cohort law}
  Let $N_t^{g,a}$ denote resident persons of sex $g$ and single year of age $a$.
  The annual law is
  \[
    N_{t+1}=A_{t+1}N_t+f_{t+1}B_{t+1}+M_{t+1},
  \]
  where
  \begin{itemize}
    \item $A_{t+1}$ ages existing cohorts using destination-age survival;
    \item $f_{t+1}B_{t+1}$ allocates newborn survivors by sex at age zero;
    \item $M_{t+1}$ is the age--sex net-migration flow after survival.
  \end{itemize}

  The model produces a four-year birth flow. It is spread uniformly across the
  four calendar years, while each year uses its own sex shares, survival, and
  migration inputs.

  \takeaway{Births create persons; survival and migration determine which cohorts are resident four years later.}
\end{frame}

\begin{frame}{Head formation and the household distribution}
  The head stock is a separate accounting object. In each age--sex cell,
  surviving heads are adjusted by the minimum one-way flow needed to attain the
  externally fixed headship rate $\rho^{g,a}$:
  \[
    H_{t+1}^{g,a}=\rho^{g,a}N_{t+1}^{g,a}
    =H_{t+1}^{\mathrm{surv},g,a}+F_{t+1}^{g,a}-D_{t+1}^{g,a}+H_{t+1}^{M,g,a}.
  \]

  The economic household operator supplies a raw within-age distribution
  $\widetilde G_{t+1,a}$. Its mass is rescaled to the person-law head target:
  \[
    G_{t+1,a}(x)=H_{t+1,a}
    \frac{\widetilde G_{t+1,a}(x)}{\int\widetilde G_{t+1,a}(x')\,dx'}.
  \]

  The conditional wealth, tenure, income, parity, and child-state composition
  is preserved within age; this is an accounting closure, not an estimated
  household-formation hazard.
\end{frame}

\begin{frame}{Transition equilibrium}
  A perfect-foresight transition consists of paths
  \[
    \{V_t,\pi_t,G_t,N_t,H_t,P_t,r_t,T_t\}_{t=2023}^{T}
  \]
  such that, at every date:
  \begin{enumerate}
    \item household policies $\pi_t$ solve the Bellman problem given future prices and transfers;
    \item births from $G_t$ advance persons and heads, and the household distribution is rebalanced to head mass;
    \item occupied housing equals static-elastic supply;
    \item the asset-price/rent identity holds;
    \item property-tax revenue equals the equal transfer paid to household heads.
  \end{enumerate}

  \takeaway{Prices are forward looking, while the person and household distributions move forward from the same initial condition.}
\end{frame}

\begin{frame}{The corrected steady state}
  A steady state is a constant tuple
  \[
    (N^*,H^*,G^*,\pi^*,P^*,r^*,T^*)
  \]
  satisfying all of the following simultaneously:
  \begin{itemize}
    \item household optimality and household-distribution stationarity;
    \item the person-cohort fixed point under the stated survival and migration closure;
    \item head formation and exact agreement between head mass and household mass;
    \item housing-market clearing and the stationary asset-price/rent identity;
    \item the government budget and demographic reproduction.
  \end{itemize}

  A finite path is not a steady state. Nor is a constant household distribution
  whose implied births fail to reproduce the person and head stocks.
\end{frame}

\section{Calibration}

\begin{frame}{Dated calibration}
  The model is estimated at a point along the 2007--2023 transition.

  \begin{enumerate}
    \item The 2007 fertility intercept reproduces completed fertility of 2.1 in a constructed old benchmark.
    \item Only the age marginal is reweighted to the 2007 ACS head distribution.
    \item A reduced-form fertility-preference shifter moves linearly through 2023.
    \item Census household totals and ACS head-age shares are imposed at five dates.
    \item Twelve moments, principally measured on the 2023 state, discipline eleven free parameters.
  \end{enumerate}

  \vspace{0.3em}
  The preference path is not interpreted as a causally identified shock date,
  and the imposed household bridge is not counted as fit.
\end{frame}

\section{Policy}

\begin{frame}{The rebated property tax}
  The reform permanently raises the annual property-tax rate from 1\% to 2\%
  and returns all contemporaneous revenue equally across household heads. With
  $\tau_{H,t}=4\tau^{\mathrm{ann}}_{H,t}$ in a four-year model period,
  \[
    T_t\int dG_t
    =\tau_{H,t}P_t Q_t^H,
    \qquad Q_t^H\equiv\int h_t(x)\,dG_t(x).
  \]

  \begin{itemize}
    \item \textbf{Capitalization:} the higher carrying cost tends to reduce the asset price.
    \item \textbf{User cost:} the renter price depends on both the tax rate and the next asset price.
    \item \textbf{Tenure and fertility:} down payments, owner products, and child-space choices adjust state by state.
    \item \textbf{Demographic scale:} changed births affect persons immediately and heads only with a lag.
  \end{itemize}

  This is an equilibrium incidence exercise, not a welfare comparison.

  \source{Policy contract: identical initial state, survival, migration, headship, supply elasticity, and terminal fertility preference across tax cases.}
\end{frame}

\begin{frame}{Conclusion}
  \begin{enumerate}
    \item Children raise demand for rooms when rental size and down payments constrain access to family housing.
    \item A lifecycle model is needed because fertility, tenure, saving, and old-age housing retention are jointly determined.
    \item A coherent transition must move persons first and derive household heads through survival, migration, and formation.
    \item The fitted model matches fertility stocks, tenure, and wealth more closely than housing quantities and is locally weakly identified.
    \item The accepted rebated 1\% path displays demographic momentum followed by a slow transition to a conditional fixed-migration equilibrium.
  \end{enumerate}

  \takeaway{Housing policy can affect family formation today and the scale of housing demand generations later; the two margins require one internally consistent equilibrium law.}
\end{frame}

\begin{frame}{Compact household derivation}
  The young household in the simple model solves
  \[
    \max_{c,h,n}\ c+\alpha\log h+v\log n
    \quad\text{s.t.}\quad
    c+rh+\bigl[q+a(r+z)\bigr]n=y.
  \]
  Interior first-order conditions give
  \[
    h(r)=\frac{\alpha}{r},
    \qquad
    n(r,z)=\frac{v}{q+a(r+z)},
    \qquad
    d_Y(r,z)=\frac{\alpha}{r}+a n(r,z).
  \]
  Thus
  \[
    \frac{\partial n}{\partial r}
    =\frac{\partial n}{\partial z}
    =-\frac{av}{[q+a(r+z)]^2}<0.
  \]

  The quantitative model preserves the cost channel but replaces continuous
  fertility, space, and tenure with sequential fertility, a rental cap, and a
  discrete owner ladder.
\end{frame}

\begin{frame}{Access-gap interpretation}
  For a constrained renter, divide the housing first-order condition by the
  marginal utility of consumption:
  \[
    \frac{u_h}{u_c}=r+\zeta_i^R,
    \qquad \zeta_i^R\ge0.
  \]
  For an owner, the same gap separates the down-payment and size constraints:
  \[
    \zeta_i^O=
    \underbrace{\frac{\mu_i}{\lambda_i}(1-\phi)P}_{\text{financing}}
    +\underbrace{\frac{\theta_i}{\lambda_i}}_{\text{owner size limit}}.
  \]
  If each child requires $a$ units of space, $a\zeta_i$ is the shadow child cost
  generated by limited access. The gap is a local marginal valuation, not the
  welfare gain from an entire extra room.
\end{frame}

\begin{frame}{Bellman timing}
  Within a four-year date, the household:
  \begin{enumerate}
    \item observes fertility taste shocks and, when eligible, chooses whether to wait or attempt a birth;
    \item realizes conception success and chooses renter rooms or an owner product conditional on parity;
    \item chooses consumption and next-period liquid wealth;
    \item receives income and pays rent, property tax, depreciation, and any moving cost;
    \item ages children independently, transitions income, and survives or leaves a warm-glow estate.
  \end{enumerate}

  In a transition, date-$t$ policies use the date-$(t+1)$ continuation value.
  The terminal value function and terminal next-date asset price are boundary
  conditions; they do not replace the requirement that the simulated terminal
  distribution approach the terminal fixed point.
\end{frame}

\begin{frame}{Exact annual person accounting}
  The annual law can be written cell by cell. For newborns,
  \[
    N_{t+1}^{g,0}=s_{t+1}^{g,0}f_{t+1}^gB_{t+1}+M_{t+1}^{g,0}.
  \]
  For interior ages $a\ge1$,
  \[
    N_{t+1}^{g,a}=s_{t+1}^{g,a}N_t^{g,a-1}+M_{t+1}^{g,a}.
  \]
  The terminal age is top-coded, so survivors from the penultimate and terminal
  cells both enter the last cell. Every step satisfies
  \[
    \sum N_{t+1}=\sum N_t+B_{t+1}
    -\text{newborn deaths}-\text{existing deaths}+\sum M_{t+1}.
  \]
\end{frame}

\begin{frame}{Exact household-head accounting}
  Surviving heads are aged with the same survival schedule. In each cell, the
  target is $\rho^{g,a}N_{t+1}^{g,a}$. The accounting closure uses only one
  gross adjustment direction:
  \[
    F_{t+1}^{g,a}
    =\max\{\rho^{g,a}N_{t+1}^{\mathrm{preM},g,a}
      -H_{t+1}^{\mathrm{surv},g,a},0\},
  \]
  \[
    D_{t+1}^{g,a}
    =\max\{H_{t+1}^{\mathrm{surv},g,a}
      -\rho^{g,a}N_{t+1}^{\mathrm{preM},g,a},0\},
    \qquad
    H_{t+1}^{M,g,a}=\rho^{g,a}M_{t+1}^{g,a}.
  \]
  Hence $H_{t+1}^{g,a}=\rho^{g,a}N_{t+1}^{g,a}$ exactly. Formation and
  dissolution are accounting flows, not simultaneous estimated hazards.
\end{frame}

\begin{frame}{The within-age household bridge}
  Let $\widetilde G_{t+1,a}(x)$ be the raw household transition within age $a$,
  and let $H_{t+1,a}$ be the head mass implied by persons. Then
  \[
    G_{t+1,a}(x)=
    \begin{cases}
      \displaystyle H_{t+1,a}
      \frac{\widetilde G_{t+1,a}(x)}{\int\widetilde G_{t+1,a}(x')dx'},
      & \int\widetilde G_{t+1,a}>0,\\[1.2em]
      H_{t+1,a}\,\Gamma_a(x), & \text{for an empty entrant age cell},
    \end{cases}
  \]
  where $\Gamma_a$ is the externally specified entrant template. The operation
  exposes added and removed mass separately and moves no mass across ages.

  A richer model would estimate economic states for newly formed, dissolved,
  and migrant heads rather than preserve the raw conditional composition.
\end{frame}

\begin{frame}{Steady-state demographic scale}
  With fixed terminal survival $A$, newborn-survivor vector $f$, net migration
  $m$, and headship vector $\rho$, persons satisfy
  \[
    N^*=AN^*+f\,b^*(\rho'N^*)+m,
  \]
  where $b^*$ is annual births per head implied by the household policy.

  Define the person responses
  \[
    v_B=(I-A)^{-1}f,
    \qquad
    v_M=(I-A)^{-1}m.
  \]
  Then the demographic renewal ratio is
  \[
    \mathcal R=b^*\rho'v_B,
  \]
  and, when fixed migration is positive, a finite steady state requires
  $\mathcal R<1$. Annual births and persons are
  \[
    B^*=\frac{b^*\rho'v_M}{1-\mathcal R},
    \qquad N^*=v_M+B^*v_B.
  \]
\end{frame}

\begin{frame}{Market and fiscal closure}
  At every date, occupied physical rooms satisfy
  \[
    \int h(x;\pi_t)\,dG_t(x)
    =H_{2023}\left(\frac{P_t}{P_{2023}}\right)^{\eta}.
  \]
  The rebated tax satisfies
  \[
    \tau_{H,t}P_t Q_t^H
    =T_t\int dG_t.
  \]
  At a stationary price,
  \[
    r^*=(R-1+\delta+\tau_H)P^*.
  \]

  Housing demand uses physical rooms. The owner-service premium affects utility
  but does not create additional physical supply. Conditional renter housing is
  also distinct from realized housing after the tenure choice.
\end{frame}

\begin{frame}{Historical calibration bridge}
  The 2007--2023 calibration uses five four-year dates. At each date, the
  observed household total and head-age marginal are imposed before the twelve
  maintained moments are evaluated.

  \begin{itemize}
    \item This absorbs household formation and migration that births since 2007 cannot generate by 2023.
    \item It makes the cross-section a dated state rather than a stationary distribution.
    \item Historical fertility, ownership, rooms, and price paths are holdouts unless explicitly included in the target system.
    \item The bridge is retired after 2023 and replaced by the person/head law.
  \end{itemize}

  Consequently, matching the imposed population path is an accounting identity,
  not evidence that the model explains U.S. household growth over 2007--2023.
\end{frame}

\begin{frame}{Slow terminal convergence}
  The dominant eigenvalue of the frozen-terminal four-year demographic block is
  \[
    \lambda_1=0.960754.
  \]
  It implies
  \[
    \text{half-life}=\frac{4\log(1/2)}{\log\lambda_1}\approx69\ \text{years},
    \qquad
    \lambda_1^{116}<0.01.
  \]

  Thus one percent attenuation of the slow mode requires about 116 model
  periods, or 464 years. The accepted path uses 128 four-year periods so the
  simulated state can approach the terminal fixed point rather than merely
  inherit it as a boundary condition.

  \takeaway{The long horizon reflects age-structured demographic persistence, not a forecast horizon.}
\end{frame}

\begin{frame}{Policy comparison contract}
  A quantitative 2\%--minus--1\% comparison must hold fixed:
  \begin{itemize}
    \item the fitted 2023 household state and reconstructed person/head stocks;
    \item survival, net migration, birth sex shares, and headship;
    \item the housing-supply elasticity and 2023 supply anchor;
    \item the terminal fertility preference and all calibrated household parameters;
    \item the equal-rebate rule and the definitions of housing and fiscal residuals.
  \end{itemize}

  The comparison then separates the immediate response of prices, rents,
  tenure, and births from the delayed response of resident persons and
  household heads. No migration response or welfare ranking is part of this
  contract.
\end{frame}

\begin{frame}{Interpretive boundaries}
  \begin{itemize}
    \item The first-birth rooms contrast is descriptive and has a non-flat prepath.
    \item The active calibration is locally weakly identified in the bequest block.
    \item Fixed 2023 headship is an accounting closure, not a structural formation model.
    \item Fixed 2100 migration in levels is essential to the positive terminal population scale.
    \item Static-elastic supply omits durable construction and asymmetric decline dynamics.
    \item One national market omits spatial sorting and local migration.
    \item The policy exercise reports equilibrium incidence, not welfare.
  \end{itemize}

  These limitations narrow the interpretation without changing the exact
  accounting distinction between persons, heads, and household economic states.
\end{frame}
