\begin{frame}{Housing, Fertility, and Tenure Choice}
\textbf{Children increase how much physical space a household needs.}
Access to that space depends on tenure and wealth:
\begin{itemize}\setlength\itemsep{0.6em}
  \item Rental housing can limit the size of the home a family can occupy.
  \item Ownership expands the housing menu but requires liquid wealth for a down payment.
  \item Older owners may retain family-sized homes after their children leave.
  \item Young families and older households compete for housing: its allocation across generations matters for fertility.
\end{itemize}

\vfill
How much does access to family-sized housing affect fertility?\\
Which housing constraints matter, and for whom?
\end{frame}

\begin{frame}{This Paper}
Need a framework where fertility, tenure, housing size, saving, and population are jointly determined.

\vfill
\alert{Model:} overlapping generations with fertility, tenure, and saving
\begin{itemize}
  \item Income heterogeneity, earnings risk, and bequests.
  \item Rental size limits, down payments, and housing transaction costs.
  \item Childbirth raises housing needs and changes future housing demand.
\end{itemize}

\vfill
\alert{Empirics:} housing around births, birth timing, and wealth by age

\vfill
\alert{Quantification:} calibrate the model and study the demographic transition
\begin{itemize}
  \item Discipline the initial economy and the decline in fertility preferences.
  \item Compare housing policies from the same inherited transition state.
\end{itemize}
\end{frame}

\section{Quantitative Model}

\begin{frame}[plain,c,noframenumbering]{}
\centering
{\Large\textbf{Quantitative Model}}
\end{frame}

\begin{frame}{Environment}
\small
\begin{itemize}\setlength\itemsep{0.55em}
  \item Overlapping generations of households in one competitive housing market.
  \begin{itemize}
    \item Rental space is size-limited; ownership offers larger homes but requires a down payment.
  \end{itemize}
  \item Households age from $a=0$ to $J$, survive with probability $s_a$, and retire at $a_R$.
  \begin{itemize}
    \item Birth attempts are possible during a fertile window $[a_F^0,a_F^1]$.
    \item Children depend on their parents until they mature.
  \end{itemize}
  \item Households differ in liquid wealth $b$, inherited owned housing $h$, and persistent earnings $z$.
  \begin{itemize}
    \item Lifetime parity $n$ and dependent children $m$ describe their family composition.
  \end{itemize}
  \item Households choose birth attempts, tenure, housing size, consumption, and saving.
\end{itemize}
\end{frame}

\begin{frame}{Preferences}
\small
Households value consumption $c$, housing services $s$, and children at home $m$:
\[
u_t(c,s;m)
=e(m)^{\sigma-1}
\frac{\left[c^{\alpha_0}\bigl(s-\bar h(m)\bigr)^{1-\alpha_0}\right]^{1-\sigma}}
{1-\sigma}
+\psi_t m.
\]
Family size raises household needs through the equivalence scale $e(m)$ and the space requirement $\bar h(m)$.

\medskip
\begin{itemize}\setlength\itemsep{0.5em}
 \item $\alpha_0$ is the consumption share; $\sigma$ governs curvature.
 \item $\psi_t$ is the preference for children at date $t$.
 \item The equivalence scale $e(m)$ \cit{(Scholz et al., 2006)} is increasing and concave.
 \item Its shape is fixed externally and normalized to a childless couple.
\end{itemize}
\end{frame}

\begin{frame}{Family Space}
The first child creates a discrete increase in required housing services:
\[
\bar h(m)=h_P\1\{m>0\}.
\]
$h_P$ is the housing requirement while dependent children are present.

\medskip
Housing services depend on tenure:
\[
s=\begin{cases}
h^R,&\text{renter},\\
\chi h',&\text{owner},
\end{cases}
\qquad \chi>0.
\]
The multiplier $\chi$ converts owned rooms into housing services; $\chi>1$ gives an ownership premium.

\vfill
Child needs disappear as children mature; lifetime births remain in the state.
\end{frame}

\begin{frame}{Earnings and Bequests}
\small
Gross earnings $y_t^g(a,z)$ combine an age profile and persistent income risk \cit{(De Nardi, 2004)}. Earnings state $z$ follows a Markov process; households receive pensions after working life.

\medskip
At death, households value estate wealth $w^e$:
\[
\mathcal B(w^e)
=\theta_0\frac{(\theta_1+\max\{w^e,0\})^{1-\sigma}}{1-\sigma}.
\]
$\theta_0$ governs the bequest motive; $\theta_1$ governs its sensitivity to wealth.

\begin{itemize}\setlength\itemsep{0.5em}
  \item Estate wealth includes liquid assets and housing wealth.
  \item Bequests affect saving and old-age housing retention.
  \item Entrant wealth is externally distributed; the model is not fully dynastic.
\end{itemize}
\end{frame}

\begin{frame}{Pensions}
\small
A fixed payroll tax $\tau^{SS}$ finances pensions $\varpi_t d(z)$, with retirement at age $a_R$:
\[
\tau^{SS}\int_{a<a_R}y_t^g(a,z)\,dG_t
=\varpi_t\int_{a\geq a_R}d(z)\,dG_t.
\]
The pension $\varpi_t$ balances the budget using the actual household distribution $G_t$. The benefit multiplier $d(z)$ equals one for common benefits.

\medskip
Disposable earnings and pension income are:
\[
y_t^d(a,z)=
\begin{cases}
(1-\tau^{SS})y_t^g(a,z),&a<a_R,\\
\varpi_t d(z),&a\geq a_R.
\end{cases}
\]
Households anticipate the pension path. Earnings and pensions are measured over the model period.

\vfill
Property-tax rebates $T_t$ have a separate budget.
\end{frame}

\begin{frame}{Housing and Tenure}
Rental housing is continuous up to a size limit; owned housing comes in discrete sizes:
\[
h^R\in[0,\bar h^R],\qquad
h'\in\{0\}\cup\mathcal H^O,\qquad h'h^R=0.
\]
A household either rents or owns, and the largest owned homes exceed the rental limit.

\medskip
\begin{itemize}\setlength\itemsep{0.65em}
 \item Buying requires a down payment of $(1-\phi)P_t h'$.
 \item $P_t$ is the purchase price per room; $\phi$ is the financed share.
 \item Selling incurs a proportional transaction cost $\psi^s$.
 \item Borrowing is secured by housing; the debt limit tightens with age.
\end{itemize}
\end{frame}

\begin{frame}{Budget Constraints}
\small
Changing inherited owner size $h$ to $h'$ changes liquid resources by
\[
\mathcal T_t(h,h')=\1\{h'\neq h\}
\left[(1-\psi^s)P_t h-P_t h'\right].
\]
Housing transactions change liquid wealth before financial returns accrue.

\medskip
For renters and owners, respectively,
\[
\begin{aligned}
c+r_t h^R+b'&=R_b\bigl[b+\mathcal T_t(h,0)\bigr]+y_t^d(a,z)+T_t,\\[0.5em]
c+(\delta_H+\tau_t^p)P_t h'+b'
&=R_b\bigl[b+\mathcal T_t(h,h')\bigr]+y_t^d(a,z)+T_t.
\end{aligned}
\]
$R_b$ is the gross financial return per period, $r_t$ rent per room, $\delta_H$ depreciation, $\tau_t^p$ the property-tax rate, and $T_t$ a transfer per household head.

\vfill
Households carry net liquid wealth $b$ into the period and choose saving $b'$.
\end{frame}

\begin{frame}{Sequential Fertility and Child Aging}
\small
\begin{itemize}\setlength\itemsep{0.35em}
  \item During fertile ages, a household below terminal parity chooses whether to attempt the next birth.
  \item An attempt succeeds with age-specific probability $\pi_a$.
  \item A successful birth:
  \[
    n'=n+1,\qquad m'=m+1,
  \]
  and the first successful birth pays the one-time utility cost $F_1$.
  \item Fertility taste shocks have scale $\kappa_1$ on the first-birth margin and $\kappa_C$ on continuation margins.
  \item Each dependent child matures independently; the state tracks their count $m$, while lifetime parity $n\in\{0,1,2,3+\}$ remains a stock.
\end{itemize}

\vfill
A birth changes the household's space needs immediately, but its demographic effects propagate across cohorts.
\end{frame}

\begin{frame}{Housing Supply and Prices}
\small
Physical housing supply responds to the purchase price:
\[
H_t^s=H_0\left(\frac{P_t}{\bar P}\right)^\eta.
\]
$H_0$ is supply at reference price $\bar P$; $\eta$ is the supply elasticity.

\medskip
Rent and purchase prices satisfy the asset-pricing condition:
\[
r_t+P_{t+1}=(R_b+\delta_H+\tau_t^p)P_t.
\]
The current rent depends on both the carrying cost and the anticipated resale price.

\vfill
Housing is supplied separately at each date; there is no construction stock in this specification.
\end{frame}

\only<1>{%
\small
At date $t$ and age $a$, write the household's inherited state as
\[
x=(b,h,z,n,m).
\]
$b$ is liquid wealth, $h$ owned housing ($h=0$ for renters), $z$ the earnings state, $n$ lifetime parity, and $m$ dependent children.

\medskip
$V_{t,a}(x)$ is value before fertility choice; $\mathcal V_{t,a}(x)$ is value after the birth outcome, before housing choice.

\medskip
A successful birth changes the state to $x^+=(b,h,z,n+1,m+1)$. Households then choose housing and saving for the realized family size.
}

\only<2>{%
\footnotesize
After the birth outcome, housing and saving solve
\[
\begin{aligned}
\mathcal V_{t,a}(b,h,z,n,m)
=\E_{\varepsilon_t^H}\max_{h',h^R,b'}
\Bigl\{&u_t(c,s;m)+\varepsilon_t^H(h')\\
&+\beta s_a\E\!\left[V_{t+1,a+1}(b',h',z',n,m')\mid z,m\right]\\
&+\beta(1-s_a)\E_t[\mathcal B(w^e_{t+1})]\Bigr\}.
\end{aligned}
\]
\begin{itemize}\setlength\itemsep{3pt}
  \item Choices satisfy budgets, the rental size cap, down-payment requirements, and age-dependent debt limits.
  \item Expectations integrate next period's earnings $z'$ and dependent-child count $m'$.
  \item $\beta$ discounts the future, $s_a$ is survival, and $\kappa_H$ is the scale of housing tastes.
\end{itemize}
}

\only<3>{%
\footnotesize
Given conception success $\pi_a$, waiting and attempting have values
\[
W_{t,a}^0(x)=\mathcal V_{t,a}(x),\qquad
W_{t,a}^A(x)=\pi_a\!\left[\mathcal V_{t,a}(x^+)-F_1\1\{n=0\}\right]
+(1-\pi_a)\mathcal V_{t,a}(x).
\]
The value before fertility tastes and the attempt probability are
\[
V_{t,a}(x)=\E_{\varepsilon^F}\max\{W_{t,a}^0(x)+\varepsilon_0^F,\ W_{t,a}^A(x)+\varepsilon_A^F\},
\]
\[
\pi^F_{t,a}(x)=\frac{\exp\{W_{t,a}^A(x)/\kappa(n)\}}
{\exp\{W_{t,a}^A(x)/\kappa(n)\}+\exp\{W_{t,a}^0(x)/\kappa(n)\}},
\qquad \kappa(n)=\begin{cases}\kappa_1,&n=0,\\\kappa_C,&n\geq1.\end{cases}
\]
Expected births are $\pi_a\pi^F_{t,a}(x)$. Attempts are unavailable outside fertile ages or at terminal parity.
}
\end{frame}

\begin{frame}{Population and Households}
Births change future housing demand by changing the number and age of households.

\medskip
\begin{itemize}\setlength\itemsep{0.65em}
  \item $N_t$: resident persons, tracked by age and sex.
  \item $H_t$: household heads, obtained from persons using age--sex headship rates.
  \item $G_t$: the distribution of household wealth, earnings, tenure, and children.
\end{itemize}

\medskip
Births enter the population immediately. Survival, migration, and household formation determine when they add to housing demand.

\vfill
\hfill\hyperlink{person-accounting}{\beamergotobutton{Population accounting}}
\end{frame}

\begin{frame}{Equilibrium}
\small
An equilibrium is a sequence of policy functions, value functions, household distributions, populations, prices, transfers, and pensions such that:
\begin{enumerate}\setlength\itemsep{0.35em}
  \item Given prices, fiscal paths, and expectations, policies solve the household's problem and values give the associated lifetime utilities.
  \item Annual population accounting and fixed headship rates satisfy:
  \[
  N_{y+1}=A_{y+1}N_y+f_{y+1}B_{y+1}+M_{y+1},
  \qquad H_y^{g,a}=\rho^{g,a}N_y^{g,a}.
  \]
  Fertility policies determine $B$; $A$ ages survivors, $fB$ counts surviving newborns, and $M$ is net migration.
  \item From the initial state, distributions follow policies, idiosyncratic shocks, and demographic entry and exit, with $\int dG_t=H_t$, the head count over represented adult ages.
  \item Housing clears, $\int q_t\,dG_t=H_t^s$, with $q_t$ measured in physical rooms. Asset pricing and both fiscal budgets hold.
\end{enumerate}
\end{frame}

\begin{frame}{Calibration Strategy}
\small
Discipline household behavior in an approximate pre-2007 stationary economy, then hold structural parameters fixed along the demographic transition.

\medskip
\begin{enumerate}\setlength\itemsep{0.65em}
 \item \textbf{External inputs.} Fix the income process, survival and conception probabilities, housing menus, equivalence scale, and housing-supply elasticity.
 \item \textbf{Initial calibration.} Jointly match fertility, housing, and wealth moments by minimum distance. Normalize $\psi_0$ to model completed fertility of 2.1.
 \item \textbf{Historical transition.} Initialize household masses in 2007 and choose subsequent fertility-preference levels to match dated fertility rates.
\end{enumerate}

\vfill
The initial fit disciplines behavior; the historical fit disciplines changes in the preference for children.
\end{frame}

Compare model moments with the same age windows, conditioning events, and time units used in the data.

\medskip

 \item \textbf{Housing:} ACS 2005/2006, pooled across 42 metros; match room top-codes, age ranges, and resident-child groups. The model uses dependent counts as the child-group proxy.
 \item \textbf{Wealth:} PSID 2003/2005; match weighted wealth/earnings and old-age wealth/income dispersion. Bequests/wealth is an external flow restriction.

\end{itemize}

\begin{frame}{From the Initial Economy to 2023}
\small
The historical exercise separates inherited demographics from changes in fertility preferences.

\medskip
\begin{enumerate}\setlength\itemsep{0.55em}
 \item \textbf{2007 starting point.} Impose observed age masses, preserving the initial stationary distribution of household states within each age.
 \item \textbf{Preference innovations.} At each decision date, households observe a new preference level and believe it will remain permanent.
 \item \textbf{Conditional equilibrium.} Solve the anticipated price and pension paths jointly with household choices and population accounting. Realize the first four-year period.
 \item \textbf{Next observation.} Carry forward wealth, homes, household distributions, and birth cohorts. Choose the next preference level to fit the next fertility window.
\end{enumerate}

\vfill
The pre-2023 population bridge uses observed demographics. Later preference innovations are unanticipated; the final level remains fixed after 2023.
\end{frame}

\begin{frame}{Policy Along the Transition}
At date $t_p$, introduce a housing reform after the fertility-preference decline has begun.

\medskip
Both paths start from the same state immediately before that announcement:
\[
(G_{t_p},N_{t_p},H_{t_p})^{\rm reform}
=(G_{t_p},N_{t_p},H_{t_p})^{\rm baseline}.
\]
The inherited liquid assets, homes, and age distributions are therefore identical.

\begin{itemize}\setlength\itemsep{0.5em}
 \item Keep preferences, survival, migration, and the entry rule common.
 \item Preserve the housing-supply schedule except when it is the policy instrument.
 \item Solve a new anticipated price and allocation path after the announcement.
\end{itemize}
\end{frame}

\begin{frame}{The Rebated Property Tax}
\small
Consider a permanent property-tax increase with contemporaneous revenue returned equally to household heads:
\[
T_t\int dG_t=\tau_t^p P_t Q_t^H,
\qquad Q_t^H=\int q_t(x)\,dG_t(x).
\]
Here $q_t(x)$ is occupied physical housing, including rented rooms; $\tau_t^p$ is the tax rate per model period.

\begin{itemize}\setlength\itemsep{0.5em}
 \item \textbf{Capitalization:} changes in carrying costs affect asset prices.
 \item \textbf{Access:} prices and transfers change down-payment affordability.
 \item \textbf{Household choices:} tenure, housing size, saving, and fertility adjust.
 \item \textbf{Demography:} different births change future cohorts and housing demand.
\end{itemize}

\vfill
Compare paths under the same rebate rule, with a common inherited state and supply schedule.
\end{frame}

\begin{frame}{Conclusion}
\begin{itemize}\setlength\itemsep{0.85em}
 \item Children raise demand for rooms; rental size and down payments govern access to that space.
 \item Tenure, saving, and fertility interact over the household lifecycle.
 \item Births change population and housing demand with long delays.
 \item Housing policy changes choices along an ongoing demographic transition.
\end{itemize}

\vfill
\textbf{The policy comparison is between paths starting from the same inherited economy.}
\end{frame}

\begin{frame}{Steady-State Equilibrium}
\small
A steady state is a constant allocation, population structure, and price:
\[
(N^*,H^*,G^*,\pi^*,P^*,r^*,T^*,\varpi^*).
\]
Households optimize, persons and household distributions reproduce themselves, housing clears, and the fiscal rule holds.

\medskip
\begin{itemize}\setlength\itemsep{0.65em}
 \item With no migration, births must replace deaths in a constant population.
 \item With migration, births and net inflows together offset deaths.
 \item A constant continuation value alone does not make a finite path stationary.
\end{itemize}
\end{frame}

\begin{frame}[label=person-accounting]{The Person-Cohort Law}
\small
Let $N_t^{g,a}$ denote resident persons of sex $g$ and single year of age $a$.

\medskip
The annual person law is
\[
N_{t+1}
=
A_{t+1}N_t+f_{t+1}B_{t+1}+M_{t+1}.
\]

\begin{itemize}\setlength\itemsep{0.5em}
  \item $A_{t+1}$ ages existing cohorts using destination-age survival.
  \item $f_{t+1}B_{t+1}$ allocates newborn survivors by sex at age zero.
  \item $M_{t+1}$ is the age--sex net-migration flow after survival.
  \item Model-period births are assigned to calendar cohorts before the annual law is iterated.
\end{itemize}

\vfill
Births and household heads remain distinct objects throughout the transition.
\end{frame}

\begin{frame}{The Household-Head Law}
\small
Fixed age--sex headship rates $\rho^{g,a}$ map persons into the target head stock:
\[
H_{t+1}^{g,a}
=
\rho^{g,a}N_{t+1}^{g,a}
=
H_{t+1}^{\mathrm{surv},g,a}
+F_{t+1}^{g,a}
-D_{t+1}^{g,a}
+H_{t+1}^{M,g,a}.
\]

The household operator supplies a raw within-age distribution
$\widetilde G_{t+1,a}$. Its mass is rescaled to the person-law head target:
\[
G_{t+1,a}(x)
=
H_{t+1,a}
\frac{\widetilde G_{t+1,a}(x)}
{\int\widetilde G_{t+1,a}(x')\,dx'}.
\]

\begin{itemize}\setlength\itemsep{0.4em}
  \item $F$ and $D$ are the minimum one-way formation or dissolution flow needed to attain $\rho^{g,a}$.
  \item Wealth, tenure, income, parity, and child-state composition are preserved within age.
  \item This is an accounting closure, not an estimated household-formation hazard.
\end{itemize}
\end{frame}

\begin{frame}{Exact Annual Person Accounting}
\small
Let $s^{g,a}$ be survival and $f^g$ the birth sex share. For newborns,
\[
N_{t+1}^{g,0}
=
s_{t+1}^{g,0}f_{t+1}^{g}B_{t+1}
+
M_{t+1}^{g,0}.
\]
For interior ages $a\geq1$,
\[
N_{t+1}^{g,a}
=
s_{t+1}^{g,a}N_t^{g,a-1}
+
M_{t+1}^{g,a}.
\]
The terminal age is top coded, so survivors from the penultimate and terminal
cells both enter the last cell. Every step satisfies
\[
\sum N_{t+1}
=
\sum N_t+B_{t+1}
-\text{newborn deaths}
-\text{existing deaths}
+\sum M_{t+1}.
\]
\end{frame}

\begin{frame}{The Within-Age Household Bridge}
Let $\widetilde G_{t+1,a}(x)$ be the raw household transition within age $a$,
and let $H_{t+1,a}$ be the head mass implied by persons. Then
\[
G_{t+1,a}(x)
=
\begin{cases}
\displaystyle
H_{t+1,a}
\frac{\widetilde G_{t+1,a}(x)}
{\int\widetilde G_{t+1,a}(x')\,dx'},
&
\int\widetilde G_{t+1,a}>0,
\\[1.2em]
H_{t+1,a}\Gamma_a(x),
&
\text{for an empty entrant age cell},
\end{cases}
\]
where $\Gamma_a$ is the externally specified entrant template. The operation
exposes added and removed mass separately and moves no mass across ages.

A richer model would estimate economic states for newly formed, dissolved,
and migrant heads rather than preserve the raw conditional composition.
\end{frame}

\begin{frame}{Housing-Market Clearing}
Let $q_t(x)$ denote the physical housing occupied after tenure choice. Market clearing requires
\[
Q_t^H=\int q_t(x)\,dG_t(x)
=H_0\left(\frac{P_t}{\bar P}\right)^\eta.
\]
For renters, $q_t=h_t^R$; for owners, $q_t=h_t'$. Choice probabilities are included when aggregating.

\medskip
\begin{itemize}\setlength\itemsep{0.65em}
 \item The owner-service premium changes utility, not physical supply.
 \item Conditional renter policies enter demand only when renting is chosen.
 \item The same supply schedule applies to baseline and tax-reform paths.
\end{itemize}
\end{frame}

\begin{frame}{Perfect-Foresight Solution}
\begin{enumerate}\setlength\itemsep{0.7em}
 \item Guess the price and pension paths and compute rents from asset pricing.
 \item Solve households backward using date-specific continuation values.
 \item Advance household and population distributions from the inherited state.
 \item Update prices and pensions until housing clears and the payroll budget balances at every date.
 \item Extend the horizon and check that the measured outcomes are stable.
\end{enumerate}

\vfill
A stationary terminal value supplies a boundary condition; it does not establish stationarity of the simulated endpoint.
\end{frame}
