% Closed-population algorithms used by the original household-queue experiment.
% These frames describe the solver and acceptance checks, not convergence results.
\begin{frame}[label=algo-ss]{Steady-State Solution}
\titlelink{transition-short}{Back}
Solve for the house price $P$, pension $\varpi$, and rebate $T$, given parameters and a fixed fertility preference.

\medskip
\begin{enumerate}\setlength\itemsep{0.55em}
 \item Guess $(P,\varpi,T)$; derive rent $r$ from the user cost.
 \item Solve households backward over age; compute choice probabilities.
 \item Construct the stationary household distribution. Set household mass equal to housing supply divided by mean housing per household.
 \item Compute three residuals: housing excess demand, the pension budget, and the rebate budget. The loss is the largest residual in absolute value.
 \item Update $(P,\varpi,T)$ and repeat until the loss is below tolerance. Then apply the population law for one period and verify that the distribution reproduces itself.
\end{enumerate}
\end{frame}

\begin{frame}[label=algo-transition]{Transition Solution}
\titlelink{transition-short}{Back}
Solve for the paths $(P_t,\varpi_t,T_t)_{t\ge t_0}$, given inherited households and a preference believed permanent.

\medskip
\begin{enumerate}\setlength\itemsep{0.4em}
 \item Solve the new steady state; it supplies the terminal value function.
 \item Guess the three paths; derive rents from the user cost.
 \item Solve households backward in calendar time from the terminal value.
 \item Advance households and entry queues forward from the inherited state; aggregate housing demand and taxes at each date.
 \item Compute the three residuals at every date. The loss is the largest residual across dates.
 \item Update the paths jointly and repeat until the loss is below tolerance. Extend the horizon to check that early responses are stable.
\end{enumerate}
\end{frame}
