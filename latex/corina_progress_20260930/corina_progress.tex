% 2: Household choices and lifecycle risks in the retained model.
\begin{frame}{Household decisions}
\small
Overlapping generations of households differ in age, earnings, wealth, housing tenure, and family size.
\medskip
\begin{itemize}\setlength\itemsep{7pt}
 \item During fertile ages, households choose whether to attempt a birth; success becomes less likely with age.
 \item They choose consumption, saving, housing size, and whether to rent or own; renters pay rent, while owners make a down payment and face selling costs.
 \item Persistent earnings risk and borrowing limits shape saving; a first-birth fixed cost and taste shocks shape fertility.
 \item Households retire, face mortality risk, and value bequests.
\end{itemize}

\end{frame}

% 3: Household preferences under the parenthood-only Stone--Geary specification.
\begin{frame}{Children and housing needs}
\small
The first child requires basic space at home; each additional child raises the goods needs of a larger family.
\[
\begin{aligned}
u(c,s;m)&=\frac{\left[\dfrac{c^{\alpha_0}s^{1-\alpha_0}}{e(m)}\right]^{1-\sigma}}{1-\sigma}
 +\psi m^{1-\gamma},\\
e(m)&=\left(\frac{2+0.7m}{2}\right)^{0.7},\qquad
s=\eta_{\tau}\left[h-h_P\mathbf 1\{m>0\}\right],\\
\eta_{R}&=1,\qquad \eta_{O}=\chi.
\end{aligned}
\]
Here $h$ is rooms, $m$ is children at home, and $\tau\in\{R,O\}$ denotes tenure. Parents require $h>h_P$; $\chi$ raises housing services for owners. The equivalence scale $e(m)$ increases family consumption needs. Child benefits are zero at $m=0$; $\psi$ sets their level and $\gamma$ controls diminishing marginal benefits.
\end{frame}

% 4: Demographic transitions and market/fiscal closure.
\begin{frame}{Demographic and market equilibrium}
\small
Children leaving home and new adult households entering the economy are separate transitions.
\medskip
\begin{itemize}\setlength\itemsep{5pt}
 \item Children leave dependency independently; adult entry follows earlier births with a lag.
 \item Positive estates finance entrant assets; any remainder leaves the economy.
 \item Stationary calibration normalizes population to one: the house price satisfies birth renewal, and housing clearing determines $H_0$.
 \item A balanced payroll tax finances equal retiree pensions.
\end{itemize}

\end{frame}

\begin{frame}{Why credit does not reach the birth decision}
\small
\begin{itemize}\setlength\itemsep{6pt}
 \item Rent per room equals the owner's per-room user cost. Credit changes who holds the house, not the price of the space a child requires.
 \item The owner premium $\chi$ applies equally to parents and to childless households, so easier ownership raises the value of both alternatives in the birth decision.
 \item The rental cap binds often but costs little. Among renting parents, 60\% are at the six-room cap; without the cap the median would rent 7.3 rooms.
 \item Being held at the cap costs parents 2.3\% of period consumption, against 1.6\% for capped childless renters.
 \item The parent housing requirement is the same with one, two or three children, so it never pushes a family above the rental cap.
\end{itemize}
\vfill
\footnotesize Cap calculations hold each household's saving fixed and use saved policies at the quarter-saving calibration.
\end{frame}

% 13: The purchase-timing rules.
\begin{frame}{Credit timing}
\small
With $\phi=0.8$, a purchase requires a 20\% down payment. Purchase timing determines which resources count toward it. The soft rule includes current-period income in the purchase screen; the other rules use:
\[
\begin{aligned}
\text{Hard:}\quad&A\ge (1-\phi)Q,\\
\text{Quarter-saving:}\quad&A+\tfrac14\left[b'-(A-Q)\right]\ge (1-\phi)Q.
\end{aligned}
\]
Here $A$ is financial wealth plus net proceeds from selling the previous home, $Q$ is the cost of the chosen home, and $b'$ is the end-of-period financial balance. Net saving is $b'-(A-Q)$.
\par\medskip
For a home costing 100 with 10 available at closing, the hard rule blocks the purchase. The quarter-saving rule meets the 20-unit equity requirement if net saving is 40: $10+0.25\times40=20$.
\par\medskip
This approximates saving before purchase within a four-year period. Under quarter-saving, income helps only if saved. Hard and quarter-saving also impose an end-of-period net-estate floor for buyers; this is an additional change from soft.
\end{frame}
