% ---------------------------------------------------------------- roadmap
\begin{frame}{Since May}
One-market lifecycle model of housing and fertility. What changed:
\begin{itemize}
  \item \textbf{Sequential fertility}: birth-by-birth attempts under declining fecundity; children ever born and children at home tracked separately
  \item \textbf{Idiosyncratic income risk}: persistent after-tax earnings process, set externally
  \item \textbf{One housing market}: the two-location structure is set aside for the national question
\end{itemize}
\end{frame}

\section{Model}

\begin{frame}[plain,c,noframenumbering]
\centering
{\Large\textbf{Model}}
\end{frame}

% ---------------------------------------------------------------- environment
\begin{frame}{Environment}
\begin{itemize}
  \item One national housing market; stationary equilibrium with household population $S$
  \item Households live $J$ four-year periods from age 18; retirement, then survival risk
  \item Two child states: $n$ children ever born, $m$ children currently at home
  \begin{itemize}
    \item Children at home raise consumption and housing needs
    \item Each child matures stochastically, leaves the household, and enters the economy as a young adult
  \end{itemize}
  \item Idiosyncratic persistent earnings risk; wages and the interest rate are exogenous
  \item Households rent, or own a discrete house size on a ladder financed with a mortgage
  \item Government: property tax rebated lump sum; payroll tax funds a pension
  \item New households enter as young renters: matured children plus an outside inflow
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- preferences
\begin{frame}{Preferences}
\small
\setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}
CRRA utility per equivalent adult over a consumption--housing bundle:
\[
u(c,\mathbf h;m)
= \frac{\big(x/e(m)\big)^{1-\sigma}}{1-\sigma} + \psi_{\mathrm{child}}\,m,
\qquad
x = c^{\alpha}\big(\mathbf h-\underline h\, m\big)^{1-\alpha}.
\]

\textbf{Equivalence scale} (Citro--Michael 1995; Scholz--Seshadri--Khitatrakun 2006):
\[
e(m) = \left(\frac{2+0.7\,m}{2}\right)^{0.7}
\]
relative to a childless couple: each child at home raises the resources needed to reach the same utility.

\medskip
\textbf{Space requirement.} Each child at home absorbs $\underline h$ rooms before housing yields services.

\medskip
\textbf{Housing services.} Owners receive a service premium:
\[
\mathbf h = \mathbf{1}\{\text{own}\}\,\chi\, h + \mathbf{1}\{\text{rent}\}\, h^R,
\qquad \chi > 1.
\]
\end{frame}

\begin{frame}{Bequests}
Warm-glow utility at death over terminal wealth:
\[
\mathcal B(\widetilde b)
= \theta_0\,\frac{(\theta_1 + \widetilde b)^{1-\sigma}}{1-\sigma},
\qquad
\widetilde b = b + p h .
\]
\begin{itemize}
  \item $\theta_0$ governs the strength of the bequest motive; $\theta_1$ makes bequests a luxury
  \item Estates are valued by parents but transfer to no one: no intergenerational wealth flow
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- fertility
\begin{frame}{Fertility and Child Maturation}
Between ages 18 and 45, households decide each period whether to try for a child --- a discrete choice with taste shocks, with separate scales $\kappa_f$ before and after the first child.

\medskip
Trying yields a birth with the conception probability
\[
\pi(\mathrm{age}) = 1 - \omega_1\, e^{\omega_2(\mathrm{age}-18)},
\qquad \pi = 0 \text{ after } 45,
\]
which falls from $0.98$ at 18 to $0.62$ at 40: postponed births are increasingly likely never to happen.

\medskip
Each child lives at home for 18 years on average (stochastic maturation), raising the equivalence scale and the space requirement; matured children enter the economy as young households.
\end{frame}

% ---------------------------------------------------------------- income
\begin{frame}{Earnings and Institutions}
Idiosyncratic persistent earnings risk over the lifecycle:
\[
\log y_{j+1} = \rho \log y_j + \varepsilon_{j+1},
\qquad \varepsilon \sim N(0,\sigma_\varepsilon^2),
\]
discretized and combined with an age profile of earnings.
\begin{itemize}
  \item Earnings are after tax: a progressive retention function applied to gross earnings
  \item Retirees receive a stationary pension funded by a payroll tax
  \item Property tax on owners, rebated as a lump-sum transfer
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- housing
\begin{frame}{Housing, Tenure, and Budget Constraints}
\small
\setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}
Households rent a continuous quantity up to a cap, or own a discrete size on a ladder:
\[
h^R \in (0, \bar h^R], \qquad h \in \{H_1,\dots,H_K\}.
\]
\textbf{Tenure segmentation:} $\max H_k > \bar h^R$, so family-sized space is reachable only through ownership.

\smallskip
\textbf{Renters}
\[
c + b' + r\,h^R = (1+q)\,b + y(j),
\qquad b' \ge 0 .
\]
\textbf{Owners}, with prior stock $h_{-1}$
\[
c + b' + (\delta+\tau_H)\,p h
+ \mathbf{1}\{h \ne h_{-1}\}\big[p h - (1-\psi)\,p h_{-1}\big]
= (1+q)\,b + y(j),
\qquad b' \ge -\phi\, p h .
\]
\begin{itemize}
  \item $q$: interest rate; $\delta$: maintenance; $\tau_H$: property tax; $\psi$: transaction cost; $1-\phi$: down-payment share
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- supply/equilibrium
\begin{frame}{Housing Supply and Market Clearing}
Housing supply slopes upward in levels, and rents map into prices through user cost:
\[
H^S = H_0\left(\frac{r}{\bar r}\right)^{\xi},
\qquad
r = (q+\delta+\tau_H)\,p .
\]
\begin{itemize}
  \item One market-clearing condition: aggregate housing services demanded equal $H^S$
  \item Supply is in levels, not per household: a permanently larger population permanently raises rents
  \item Stationary equilibrium: the price, the transfer, and the distribution are constant, and the government budget clears
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- household problem
\begin{frame}{The Household's Problem}
Within the period, a household first decides whether to try for a child, then chooses tenure, housing, and saving:
\[
V_j(b,h,y,n,m)
= \mathbb{E}_{\varepsilon}\max_{a\in\{0,1\}}
\Big\{ W_j^{a}(b,h,y,n,m) + \varepsilon_a \Big\},
\]
\[
W_j^{a}
= \max_{\text{rent/own},\; h',\, h^{R\prime},\, b'}
\; u(c,\mathbf h; m) + \beta\,\mathbb{E}\big[V_{j+1}(b',h',y',n',m')\big]
\]
subject to the renter or owner budget and the collateral constraint; a successful try ($a=1$, probability $\pi_j$) adds a child next period.

\medskip
In old age there are no fertility decisions; survival risk and the warm glow at death close the problem.
\end{frame}

\section{Population Closure}

\begin{frame}[plain,c,noframenumbering]
\centering
{\Large\textbf{Population Closure}}
\end{frame}

% ---------------------------------------------------------------- closure problem
\begin{frame}{Demographics and Population Balance}
Completed fertility is below replacement: matured children replace about $86\%$ of exiting households. A stationary population therefore needs an inflow from outside --- as in the U.S., where net immigration now sustains population growth.

\medskip
Entrants each period are retained matured children plus an outside inflow $M$:
\[
\underbrace{S\,E_0}_{\text{entrants needed}}
\;=\;
\underbrace{R\,S\,B_0}_{\text{retained matured children}}
\;+\; M ,
\]
with $E_0$ the replacement flow per household, $B_0$ the matured-children flow, and $R$ the share who stay.

\medskip
Given a fixed inflow, the stationary population
\[
S = \frac{M}{E_0 - R\,B_0}
\]
exists and is stable: as the population shrinks, a fixed inflow becomes a growing share, and the decline stops.
\end{frame}

% ---------------------------------------------------------------- quota closure
\begin{frame}{Closing the Population}
The inflow is a quantity, not a choice: U.S.\ immigration is rationed, so marginal changes in the value of living here do not move it. $R$ and $M$ are pinned at the baseline by two observables --- the outside-origin share of young entrants ($16.9\%$, ACS) and the population normalization --- and held fixed in counterfactuals.

\medskip
Population then responds to policy through births alone. Differentiating the balance condition:
\[
\%\Delta S \;=\; \frac{1-s^{out}}{s^{out}}\;\%\Delta B_0 \;\approx\; 4.9 \times \%\Delta B_0 .
\]
Each immigrant household founds a family line that fades out at the native replacement rate; a policy that raises births makes every line last longer.

\medskip
Two qualifications. Housing damps the response: a larger population raises rents, which discourages births. And an earlier specification in which entrants weighed home values against an outside option is retired: its entry elasticity is not identified (appendix).
\end{frame}

\end{frame}

% ---------------------------------------------------------------- interpretation
\begin{frame}{Reading the Population Numbers}
The same number can be read three ways:
\begin{itemize}
  \item more households in the long run, at a fixed immigration inflow;
  \item fewer immigrants required to hold the population fixed;
  \item the share of the fertility gap closed by the policy.
\end{itemize}

\medskip
Housing feedback is present but small: the demographic arithmetic overstates the response by at most a tenth of a percentage point.

\medskip
Timing: the birth and immigration flows move immediately, while the household count converges over generations --- half of the adjustment takes roughly a century.
\end{frame}

% ---------------------------------------------------------------- open items
\begin{frame}{Open Decisions}
\begin{enumerate}
  \item \textbf{Anchoring the inflow.} Among household heads aged 18--22, recent immigrants are only $3.7\%$ --- but immigrants mostly arrive older than the model's entry age. Either anchor the inflow to total net migration in entrant equivalents (roughly $17$--$22\%$, close to the $16.9\%$ in use), or model entry at several ages
  \item \textbf{The fertility block.} The closure requires natives to replace at least $83\%$ of each cohort; the current fit sits close to that edge, and the childlessness and old-age wealth misses are limitations of the mechanism rather than of the search
  \item \textbf{Welfare.} With different populations across counterfactuals, welfare is compared per household
\end{enumerate}
\end{frame}
