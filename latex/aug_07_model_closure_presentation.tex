% ---------------------------------------------------------------- roadmap
\begin{frame}{Since May}
One-market lifecycle model of housing and fertility. What changed:
\begin{itemize}
  \item \textbf{Sequential fertility}: birth-by-birth attempts under declining fertility; children ever born and children at home tracked separately
  \item \textbf{Idiosyncratic income risk}: persistent after-tax earnings process, set externally
  \item \textbf{One housing market}: the two-location structure is set aside for the national question
\end{itemize}
\end{frame}

\section{Model}

\begin{frame}[plain,c,noframenumbering]
\centering
{\Large\textbf{Model}}
\end{frame}

% ---------------------------------------------------------------- environment
\begin{frame}{Environment}
\begin{itemize}
  \item Households live $J$ four-year periods from age 18; retirement, then survival risk
  \item Two child states: $n$ children ever born, $m$ children currently at home
  \begin{itemize}
    \item Children at home raise consumption and housing needs
    \item Each child matures stochastically, leaves the household, and enters the economy as a young adult
  \end{itemize}
  \item Idiosyncratic persistent earnings risk; wages and the interest rate are exogenous
  \item Households rent, or own a discrete house size on a ladder financed with a mortgage
  \item Government: property tax rebated lump sum; payroll tax funds a pension
  \item New households enter as young renters: matured children plus an outside inflow
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- preferences
\begin{frame}{Preferences}
Flow utility over consumption and housing services:
\[
u(c,\mathbf h;m)
= \frac{1}{1-\sigma}
\left[\frac{c^{\alpha}\big(\mathbf h-\underline h\, m\big)^{1-\alpha}}{e(m)}\right]^{1-\sigma}
+ \psi_{\mathrm{child}}\,m .
\]
\begin{itemize}
  \item \textbf{Space requirement.} Each child at home absorbs $\underline h$ rooms before housing yields services
  \item \textbf{Equivalence scale.} $e(0)=1$ and $e'>0$: a larger family needs more resources to reach the same utility
  \item \textbf{Owner premium.} $\mathbf h = \chi h$ for owners, with $\chi>1$; $\mathbf h = h^R$ for renters
\end{itemize}
\end{frame}

\begin{frame}{Bequests}
Warm-glow utility at death over terminal wealth:
\[
\mathcal B(\widetilde b)
= \theta_0\,\frac{(\theta_1 + \widetilde b)^{1-\sigma}}{1-\sigma},
\qquad
\widetilde b = b + p h .
\]
\begin{itemize}
  \item $\theta_0$ governs the strength of the bequest motive; $\theta_1$ makes bequests a luxury
  \item Estates are valued by parents but transfer to no one: no intergenerational wealth flow
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- fertility
\begin{frame}{Fertility and Child Maturation}
\begin{itemize}
  \item Between ages 18 and 45, households decide each period whether to try for a child: a discrete choice with taste shocks, with separate scales $\kappa_f$ before and after the first child
  \item Trying yields a birth with the conception probability
  \[
  \pi(\mathrm{age}) = 1 - \omega_1\, e^{\omega_2(\mathrm{age}-18)},
  \qquad \pi = 0 \text{ after } 45,
  \]
  which falls from $0.98$ at 18 to $0.62$ at 40: postponed births are increasingly likely never to happen
  \item Each child lives at home for 18 years on average, with stochastic maturation, raising the equivalence scale and the space requirement
  \item Matured children enter the economy as young households
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- income
\begin{frame}{Firms and Earnings}
\begin{itemize}
  \item A competitive firm produces with labor, so the wage is pinned down by technology and the labor market clears at any aggregate efficiency units $L$:
  \[
  Y = A\,L, \qquad w = A
  \]
  \item Households supply efficiency units inelastically: an age profile times a persistent idiosyncratic component
  \[
  \log y' = \rho \log y + \varepsilon',
  \qquad \varepsilon' \sim N(0,\sigma_\varepsilon^2)
  \]
  \item Earnings are taxed progressively, a payroll tax funds a stationary pension, and a property tax on owners is rebated lump sum
  \item With constant returns, a larger population leaves the wage unchanged: housing is the only fixed factor
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- housing
\begin{frame}{Housing, Tenure, and Budget Constraints}
\small
\setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}
Households rent a continuous quantity up to a cap, or own a discrete size on a ladder:
\[
h^R \in (0, \bar h^R], \qquad h \in \{H_1,\dots,H_K\}.
\]
\textbf{Tenure segmentation:} $\max H_k > \bar h^R$, so family-sized space is reachable only through ownership.

\smallskip
\textbf{Renters}
\[
c + b' + r\,h^R = (1+q)\,b + y(j),
\qquad b' \ge 0 .
\]
\textbf{Owners}, with prior stock $h_{-1}$
\[
c + b' + (\delta+\tau_H)\,p h
+ \mathbf{1}\{h \ne h_{-1}\}\big[p h - (1-\psi)\,p h_{-1}\big]
= (1+q)\,b + y(j),
\qquad b' \ge -\phi\, p h .
\]
\begin{itemize}
  \item $q$: interest rate; $\delta$: maintenance; $\tau_H$: property tax; $\psi$: transaction cost; $1-\phi$: down-payment share
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- supply/equilibrium
\begin{frame}{Housing Supply and Market Clearing}
Housing supply slopes upward in levels, and rents map into prices through user cost:
\[
H^S = H_0\,r^{\xi},
\qquad
r = (q+\delta+\tau_H)\,p .
\]
\begin{itemize}
  \item One market-clearing condition: aggregate housing services demanded equal $H^S$
  \item Supply is in levels, not per household: a permanently larger population permanently raises rents
  \item Stationary equilibrium: the price, the transfer, and the distribution are constant, and the government budget clears
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- household problem
\begin{frame}{The Household's Problem}
Within the period, a household first decides whether to try for a child, then chooses tenure, housing, and saving:
\[
V(b,h,y,a,n,m)
= \mathbb{E}_{\varepsilon}\max_{f\in\{0,1\}}
\Big\{ W^{f}(b,h,y,a,n,m) + \varepsilon_f \Big\},
\]
\[
W^{f}
= \max_{\text{rent/own},\; h',\, h^{R\prime},\, b'}
\; u(c,\mathbf h; m) + \beta\,\mathbb{E}\big[V(b',h',y',a+1,n',m')\big]
\]
subject to the renter or owner budget and the collateral constraint; a successful try ($f=1$, probability $\pi(a)$) adds a child next period.

\medskip
In old age there are no fertility decisions; survival risk and the warm glow at death close the problem.
\end{frame}

\section{Population Closure}

\begin{frame}[plain,c,noframenumbering]
\centering
{\Large\textbf{Population Closure}}
\end{frame}

% ---------------------------------------------------------------- closure problem
\begin{frame}{Closing the Population}
Every version of this model needs a rule for the size of the entering cohort. Three ways to supply one:
\begin{enumerate}
  \item \textbf{Balanced growth path.} Population and all quantities grow at a common rate; the model is read per capita
  \begin{itemize}
    \item Would need to handle two technical issues here: the committed housing requirement makes preferences non-homothetic, and the housing stock is fixed in levels
    \item Counterfactual anyway: the economy is not observed growing along one
  \end{itemize}
  \item \textbf{Dynamic equilibrium path.} The most general option: no stationarity, solve for the whole time path of the population and prices, markets clearing every period
  \begin{itemize}
    \item Expensive, and calibration becomes ambiguous: every observed moment must be assigned a date on the path
  \end{itemize}
  \item \textbf{Stationary equilibrium.} Compare stationary states across policies
  \begin{itemize}
    \item One long-run number per policy; calibration targets a stationary cross section
    \item Needs a rule that pins the level of the population
  \end{itemize}
\end{enumerate}
The stationary comparison is used here.
\end{frame}

% ---------------------------------------------------------------- old vs new
\begin{frame}{The Entry Rule}
Entrants: matured children who stay, plus an inflow $M$ from outside (immigration). $M$ needs a rule:
\begin{center}
$\text{earlier:}\quad M = M(V)
\qquad\qquad
\text{now:}\quad M = \bar M$
\end{center}
\smallskip
\begin{itemize}
  \item \textbf{Earlier.} Entry rises with the value $V$ of living in the economy. No moment disciplines that response, and it drives the results: population responses three to ten times larger
  \item \textbf{Now.} A fixed quota: policy moves the population only through births
  \item \textbf{Cost.} Migration cannot respond to a more attractive economy; the earlier numbers bound that channel
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- balance
\begin{frame}{Population Balance}
The economy has a mass $S$ of households: the population level. In a stationary state entrants replace exits:
\[
\underbrace{S\,E_0}_{\text{entrants needed}}
\;=\;
\underbrace{R\,S\,B_0}_{\text{matured children who stay}}
\;+\; M ,
\]
with $E_0$ exits and $B_0$ maturing children, per household; $R$ the share who stay.
\begin{itemize}
  \item The native terms scale with $S$; the inflow $M$ is a level
  \item As $S$ falls, $M$ covers a growing share of entrants and the decline stops: $S = M/(E_0 - R\,B_0)$
  \item The outside share of entrants is measured across metro areas in the ACS: $16.9\%$ of young entrant households arrive from outside, in line with national net migration relative to cohort size. Since $s^{out} = 1 - R\,B_0/E_0$, natives replace $83\%$ of each cohort, and $M$ sets the level
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- identification
\begin{frame}{Identification}
\small\linespread{1.05}\selectfont
The parameters are chosen jointly, in four economic blocks:
\begin{itemize}
  \item \textbf{Fertility.} The utility value of children $\psi_{\mathrm{child}}$ raises the return to every birth: completed fertility. The first-birth noise $\kappa_f$ spreads the age at which women start trying: mean age at first birth and first births after 30. The continuation noise $\kappa_f^{\mathrm{cont}}$ governs how readily a household that has started keeps going: childlessness
  \item \textbf{Family housing.} The child room floor $\underline h$ sets the rooms a child absorbs: the rooms added at the first child, and the rooms difference between families with three or more children and one or two. The family ownership gap has no parameter of its own and enters as an overidentifying restriction
  \item \textbf{Tenure and supply.} The owner premium $\chi$ raises the utility of owning over renting: the ownership rate. The supply scale $H_0$ sets the level of the housing stock: aggregate rooms
  \item \textbf{Saving and bequests.} The discount factor $\beta$ governs how much wealth households carry through life: net worth over earnings. The bequest strength $\theta_0$, how much is held back to leave at death: the bequest flow. The curvature $\theta_1$, how the motive strengthens with wealth: old-age p90/p50
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- policy results
\begin{frame}{Funded Property-Tax and Grant Reforms (preliminary)}
The experiment: raise the annual property tax from $1\%$ to $2\%$ of house value and return the revenue to households, either as an equal lump-sum rebate or as a grant paid at the purchase of a family-sized home. Percent changes across stationary states:
\medskip

\centering
\small
\renewcommand{\arraystretch}{0.95}
\begin{tabular}{lrrrrr}
\toprule
Policy & TFR & Births & Households & House price & Rent \\
\midrule
$2\%$ tax, lump-sum rebate & $+0.38$ & $+0.33$ & $+1.70$ & $-17.5$ & $-6.1$ \\
$2\%$ tax, purchase grant  & $+0.75$ & $+0.67$ & $+3.46$ & $-17.3$ & $-5.8$ \\
\bottomrule
\end{tabular}

\medskip
\raggedright
\begin{itemize}
  \item The tax lowers the house price, so the down payment falls by $17.5\%$; rents fall only $6\%$, because rent includes the tax itself
  \item In U.S.\ units, $+0.33$ to $+0.67\%$ of births is roughly $12$ to $24$ thousand additional babies per year
\end{itemize}
\end{frame}
