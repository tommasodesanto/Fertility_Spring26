\begin{frame}{Equilibrium closure and transition dynamics}
\textbf{Last time:} discussion of how to close population and housing markets:
\begin{itemize}
  \item closed versus open population, and the role of migration and household formation;
  \item stationary equilibrium versus a dated transition;
  \item fertility as a source of future household demand.
\end{itemize}

\medskip
\textbf{This time:} consider a closed economy that begins in a steady state and experiences a permanent fall in the preference for children:
\begin{itemize}
  \item fertility and the house price respond on impact;
  \item cohort sizes then change population and housing demand;
  \item the path may converge to a new steady state;
  \item closed and open population versions are then compared in the quantitative model.
\end{itemize}

\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{A toy economy}
\small
To fix ideas, consider a simplified economy with two household generations:
\begin{itemize}
  \item At date $t$, there are $Y_t$ young households and $O_t$ old households; total population is $S_t=Y_t+O_t$.
  \item A household lives for two periods. A young household becomes old next period; an old household exits after the current period.
  \item Young households choose consumption, adult housing, and $n_t$ \textbf{household children}. Each household child becomes one young household next period.
  \item Old households choose consumption and housing but not fertility.
  \item Both generations purchase housing services in one competitive market at price $p_t$.
\end{itemize}

\medskip
There is no migration or asset accumulation. The timing implies
\[
(Y_t,O_t)
\xrightarrow{\text{choices at }p_t}
(Y_{t+1},O_{t+1})=(n_tY_t,Y_t).
\]

\medskip
The quantitative model restores age, tenure, mortgages, saving, earnings risk, and bequests.
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Household problems}
\small
A young household earns $y_Y$ and chooses consumption $c_Y$, adult housing $h^A$, and household children $n$:
\[
\max_{c_Y,h^A,n}\left\{c_Y+\alpha\log h^A+v\log n\right\},
\qquad
c_Y+ph^A+\big[q+a(p+\omega)\big]n=y_Y.
\]
The parameter $\alpha$ values adult housing and $v$ values children. Each household child requires $q$ units of other goods and $a$ units of housing; $\omega$ is the extra unit cost of child-suitable housing. Fertility and total young-household housing demand are
\[
n(p;v)=\frac{v}{q+a(p+\omega)},
\qquad
h_Y(p;v)=\frac{\alpha}{p}+a n(p;v).
\]

An old household earns $y_O$ and has housing weight $\bar h_O$:
\[
\max_{c_O,h_O}\left\{c_O+\bar h_O\log h_O\right\},
\qquad
c_O+ph_O=y_O,
\]
which gives $h_O(p)=\bar h_O/p$. Consumption is the numeraire, and incomes are high enough for the interior choices shown above.

\textbf{A higher housing cost lowers fertility and housing demand.}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Competitive equilibrium}
\small
Housing is supplied competitively according to
\[
H^s(p)=\bar H p^\varepsilon,
\]
where $\bar H$ is the supply scale and $\varepsilon$ is the price elasticity.

Given $(Y_0,O_0)$ and a path for child preferences $\{v_t\}$, an equilibrium is a sequence of prices, household choices, and cohort masses satisfying
\[
Y_t h_Y(p_t;v_t)+O_t h_O(p_t)=H^s(p_t),
\]
and
\[
Y_{t+1}=n(p_t;v_t)Y_t,
\qquad
O_{t+1}=Y_t.
\]

Current cohort masses determine the market-clearing price; the price determines fertility and therefore the next cohort.
\end{frame}

\begin{frame}{Initial steady state}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.62\textwidth}
\centering
\includegraphics[width=\textwidth,height=0.53\textheight,keepaspectratio]{../output/model/fertility_population_housing_transition_note/steady_state_equilibrium.pdf}
\end{column}
\begin{column}{0.35\textwidth}
\footnotesize
A steady state is a constant triple $(Y^*,O^*,p^*)$. It requires
\[
n(p^*;v_0)=1,
\qquad Y^*=O^*.
\]
Replacement selects
\[
p_0=\frac{v_0-q}{a}-\omega.
\]
Housing clearing then selects $S_0=S^*(p_0)$, where
\[
S^*(p)=
\frac{2\bar H p^{1+\varepsilon}}
{\alpha+\bar h_O+a p}.
\]
Point $A$ is the initial steady state.
\end{column}
\end{columns}

\vspace{-0.75em}
\small
Along the steady-state schedule, direct differentiation gives
\[
\frac{dS^*}{dp}
=\frac{2\bar H p^\varepsilon
\big[(1+\varepsilon)(\alpha+\bar h_O)+\varepsilon a p\big]}
{(\alpha+\bar h_O+a p)^2}>0
\qquad\text{for }\varepsilon\geq0.
\]
\end{frame}

\begin{frame}{Below-replacement fertility and population dynamics}
Let $n_t=n(p_t;v_t)$ be household children per young household and $\mu_t=O_t/Y_t$ the old-to-young cohort ratio. Population can rise with $n_t<1$ whenever
\[
\mu_t<n_t<1.
\]

\begin{block}{A numerical example}
\centering
\Large
$100$ young $+$ $70$ old
$\quad\longrightarrow\quad$
$80$ young $+$ $100$ old

\normalsize
\medskip
$n_t=0.8<1$, but total households rise from $170$ to $180$.
\end{block}

Housing costs can rise as well if the aging cohort retains more housing than is released by the shrinking young cohort:
\[
(1-\mu_t)h_O(p_t)>(1-n_t)h_Y(p_t).
\]

\vspace{-0.35em}
Inherited cohorts can sustain short-run population and housing-cost growth below replacement.
\end{frame}

\begin{frame}{Dated quantitative transition}
\small
\textbf{Household problem.} Households choose fertility, tenure, housing, saving, and location. The distribution $G_t$ records their joint distribution and mass.

\medskip
\textbf{Calendar-time accounting.}
\[
G_{t+1}=\mathcal T(G_t,p_t)+E_{t+1},
\qquad
E_{t+5}=\frac{b_t^{\mathrm{child}}}{2.1}.
\]
The transition $\mathcal T$ moves households across age and economic states. Here $b_t^{\mathrm{child}}$ counts children; $2.1$ children generate one future entrant household under the replacement normalization. Entrants arrive after twenty years.

\medskip
\textbf{Housing market.} At each date, the housing cost clears demand from the current distribution:
\[
H^s(p_t)=D(G_t,p_t).
\]

\begin{itemize}
  \item \textbf{2007--2023:} household totals and age shares are imposed; the model determines states within age.
  \item \textbf{After 2023:} cohort masses evolve under the demographic law above.
\end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Calibration from 2007 to 2023}
\small
\begin{enumerate}
  \item \textbf{Replacement normalization:} completed fertility equals $2.1$.
  \item \textbf{2007 state:} match age shares and preserve economic states conditional on age.
  \item \textbf{2007--2023:} impose demographics; estimate eleven parameters from twelve moments.
  \item \textbf{Validation:} compare untargeted fertility, timing, tenure, rooms, house prices, and rents.
  \item \textbf{Post-2023:} hold preferences fixed; evolve the inherited distribution and entrant cohorts.
\end{enumerate}

\medskip
\textbf{Demographic inputs:} household totals and age shares. \textbf{Initial condition:} the 2007 distribution.
\end{frame}

\[

\begin{frame}{Rebated property tax: impact and first decade}
\footnotesize
From the same fitted 2023 distribution, permanently raise the annual property-tax rate from $1\%$ to $2\%$ and rebate the revenue equally. At each date, solve jointly for the house price and transfer under the maintained national supply elasticity, $\varepsilon_H=1.75$.

The 2031 and 2035 model dates bracket a ten-year response.
