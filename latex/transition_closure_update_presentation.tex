\begin{frame}{Equilibrium closure and transition dynamics}
\textbf{Last time:} discussion of how to close population and housing markets:
\begin{itemize}
  \item closed versus open population and the role of migration;
  \item stationary equilibrium versus a dated transition;
  \item fertility as a source of housing demand.
\end{itemize}

\medskip
\textbf{This time:} consider a closed economy that begins in a steady state and experiences a permanent fall in the preference for children:
\begin{itemize}
  \item fertility and the house price respond on impact;
  \item cohort sizes then change population and housing demand;
  \item the path may converge to a new steady state;
  \item we compare different versions in the quantitative model.
\end{itemize}

\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{A toy economy}
\small
To fix ideas, consider a simplified economy with two household generations:
\begin{itemize}
  \item At date $t$, there are $Y_t$ young households and $O_t$ old households; total population is $S_t=Y_t+O_t$.
  \item A household lives for two periods. A young household becomes old next period; an old household exits after the current period.
  \item Young households choose consumption, adult housing, and fertility $n_t$.
  \item Old households choose consumption and housing but not fertility.
  \item Both generations purchase housing services in one competitive market at price $p_t$.
\end{itemize}

\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Household problems}
\small
\begin{itemize}
  \item A young household earns $y_Y$ and chooses consumption $c_Y$, adult housing $h^A$, and fertility $n$:
\end{itemize}
\[
\max_{c_Y,h^A,n}\left\{c_Y+\alpha\log h^A+v\log n\right\},
\qquad
c_Y+ph^A+\big[q+a(p+\omega)\big]n=y_Y.
\]
\begin{itemize}
  \item $\omega$ is the housing cost of children.
  \item Implied fertility and young-household housing demand:
\end{itemize}
\[
n(p;v)=\frac{v}{q+a(p+\omega)},
\qquad
h_Y(p;v)=\frac{\alpha}{p}+a n(p;v).
\]

\begin{itemize}
  \item An old household chooses consumption and housing:
\end{itemize}
\[
\max_{c_O,h_O}\left\{c_O+\bar h_O\log h_O\right\},
\qquad
c_O+ph_O=y_O,
\]
which gives $h_O(p)=\bar h_O/p$.

\textbf{A higher housing cost lowers fertility and housing demand.}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Competitive equilibrium}
\small
Housing is supplied competitively according to
\[
H^s(p)=\bar H p^\varepsilon,
\]
where $\bar H$ is the supply scale and $\varepsilon$ is the price elasticity.

Given $(Y_0,O_0)$ and a child preference $v$, an equilibrium is a path
\[
\{p_t,c_{Y,t},h^A_t,n_t,c_{O,t},h_{O,t},Y_t,O_t\}_{t\geq0}
\]
of housing costs, household choices, and cohort masses satisfying
\[
Y_t h_Y(p_t;v)+O_t h_O(p_t)=H^s(p_t),
\]
and
\[
Y_{t+1}=n(p_t;v)Y_t,
\qquad
O_{t+1}=Y_t.
\]

Current cohort masses determine the market-clearing price; the price determines fertility and therefore the next cohort.
\end{frame}

\begin{frame}{Initial steady state}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.60\textwidth}
\centering
\includegraphics[width=\textwidth,height=0.49\textheight,keepaspectratio]{../output/model/fertility_population_housing_transition_note/steady_state_equilibrium.pdf}
\end{column}
\begin{column}{0.37\textwidth}
\footnotesize
A steady state requires
\[
n(p^*)=1,
\qquad Y^*=O^*.
\]
Here $n=1$ is replacement fertility: each young household replaces itself next period.

At replacement fertility, $p_0=(v-q)/a-\omega$.

\smallskip
Housing clearing implies steady-state total population $S^*=Y^*+O^*$:
\[
S^*(p)=
\frac{2\bar H p^{1+\varepsilon}}
{\alpha+\bar h_O+a p}.
\]
$S^*(p)$ is increasing in $p$ for $\varepsilon\geq0$; point $A$ is the initial steady state.
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Impact response}
\small
Now suppose the preference for children falls permanently. Fertility and child-related housing demand fall before cohort masses adjust.

\medskip
\begin{columns}[T,onlytextwidth]
\begin{column}{0.63\textwidth}
\centering
\includegraphics[width=\textwidth,height=0.46\textheight,keepaspectratio]{../output/model/fertility_population_housing_transition_note/partial_equilibrium_response.pdf}
\end{column}
\begin{column}{0.34\textwidth}
\footnotesize
\begin{itemize}
  \setlength\itemsep{1.0em}
  \item \textbf{Shock:} $v_0\rightarrow v_1<v_0$.
  \item \textbf{Inherited cohorts:} $Y_t=Y_0$, $O_t=O_0$, $S_t=S_0$.
  \item \textbf{Housing cost:} lower child-related demand gives $p_0\rightarrow\widetilde p$.
  \item \textbf{Fertility:} $\widetilde n=n(\widetilde p;v_1)<1$.
\end{itemize}
\end{column}
\end{columns}

\medskip
The economy moves from $A$ to $I$: fertility falls below replacement, but population has not yet adjusted.
\end{frame}

\begin{frame}{Below-replacement fertility and population dynamics}
\small
Define the old-to-young cohort ratio as $\mu_t=O_t/Y_t$. Since
\[
S_{t+1}-S_t=Y_t(n_t-\mu_t),
\]
population rises if and only if $n_t>\mu_t$. In particular, population can rise even below replacement:
\[
\mu_t<n_t<1.
\]

For example,
\[
(Y_t,O_t,n_t)=(100,70,0.8)\quad\longrightarrow\quad(Y_{t+1},O_{t+1})=(80,100),
\]
so total population rises from $S_t=170$ to $S_{t+1}=180$.

At a fixed housing cost $p_t$, aggregate housing demand rises when
\[
D_{t+1}(p_t)>D_t(p_t)
\quad\Longleftrightarrow\quad
(1-\mu_t)h_O(p_t)>(1-n_t)h_Y(p_t).
\]

Cohort composition can sustain short-run population and housing-demand growth below replacement.
\end{frame}
