def checked_plan(path, digest):
    require(sha(path)==digest, 'Continuation plan SHA drift')
    plan=json.loads(Path(path).read_text())
    anchor=json.loads(ANCHOR.read_text())
    require(anchor['status']=='selected_numerically_verified' and anchor['chain']==13 and anchor['arm']=='alternative', 'Pinned source anchor drift')
    require(plan['source_checkpoint_sha256']==sha(ANCHOR), 'Source checkpoint drift')
    require(plan['recovery_job_id']=='19141024' and plan['recovery_inventory_sha256']=='974a14e243da6a2ad0572bb9825b47ab349828f9144cbbad69e74b40a4408b22', 'Recovery identity drift')
    require(plan['parent_inventory_sha256']=='d14a39bcbb55067060ca492943f1e0067000c10733a48c3a3aff3b06e61e2afe', 'Parent identity drift')
    require(plan['parent_job_id']=='19127370' and plan['parent_array_task_job_ids']==['19127379','19127380','19127381','19127382','19127383'], 'Parent Slurm array accounting map drift')
    recovery=Path(plan['recovery_root']);parent=Path(plan['parent_root'])
    require(sha(recovery/'inventory.json')==plan['recovery_inventory_sha256'], 'Recovery inventory bytes drift')
    require(sha(parent/'inventory.json')==plan['parent_inventory_sha256'], 'Parent inventory bytes drift')
    inv=json.loads((recovery/'inventory.json').read_text())
    require(plan['target_fingerprint']==inv['target_fingerprint'] and plan['weight_fingerprint']==inv['weight_fingerprint'], 'Recovery target/weight drift')
    require(plan['target_fingerprint']==canonical(plan['target_contract']), 'Target contract fingerprint drift')
    require(plan['weight_fingerprint']==canonical(dict(base_contract=plan['target_contract'],multipliers={})), 'Weight contract fingerprint drift')
    require(plan['arms']=={'binary':1} and len(plan['bounds'])==10 and len(plan['starts'])==20, 'One-birth 20-start contract drift')
    require(plan['continuation_arm']=='binary' and len(plan['start_provenance'])==20, 'Continuation arm/provenance drift')
    require(plan['target_contract']==json.loads((parent/'source/output/model/experiments/birth_count_choice/estate_a_calibration_v1/start_plan.json').read_text())['target_contract'], 'Parent target rows drift')
    require(plan['bounds']==json.loads((parent/'source/output/model/experiments/birth_count_choice/estate_a_calibration_v1/start_plan.json').read_text())['bounds'], 'Parent parameter bounds drift')
    require(plan['source_provenance']['recovery_inventory_sha256']==plan['recovery_inventory_sha256'] and plan['source_provenance']['parent_inventory_sha256']==plan['parent_inventory_sha256'], 'Seed source provenance drift')
    for i,(point,source) in enumerate(zip(plan['starts'],plan['start_provenance'])):
        require(source['index']==i and set(point)==set(plan['bounds']), 'Start coordinate/index drift')
        require(all(math.isfinite(v) and plan['bounds'][k][0]<=v<=plan['bounds'][k][1] for k,v in point.items()), 'Start outside bounds')
        require(source['kind'] in ('verified_recovery_selected','parent_saved_feasible_checkpoint','bounded_perturbation'), 'Unknown start provenance')
    require([s['kind'] for s in plan['start_provenance']].count('verified_recovery_selected')==5, 'Five verified recovery vectors required')
    require([s['kind'] for s in plan['start_provenance']].count('parent_saved_feasible_checkpoint')==5, 'Five authenticated parent vectors required')
    require([s['kind'] for s in plan['start_provenance']].count('bounded_perturbation')==10, 'Ten bounded perturbations required')
    require(len({canonical(s) for s in plan['starts']})==20, 'Duplicate starts')
    require(min(plan['pairwise_normalized_distances'])>0.005, 'Starts are not spatially distinct')
    sys.path.insert(0,str(Path(path).resolve().parent.parent))
    from prepare_starts import audit
    require(audit(plan,recovery,parent)['status']=='all_20_starts_reauthenticated', 'Seed source reauthentication failed')
    return plan,anchor
